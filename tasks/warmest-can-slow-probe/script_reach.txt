# Baseline: random reaching. Each arm in turn moves its TCP to a random point over the work area (x 0.15-0.55 m,
# y -0.30-0.30 m, 1-25 cm above the table; 40% of targets 1-3 cm above it), with a random grip yaw and a random
# gripper opening, sometimes closing there and lifting, then holds 0-2 s. Seed: BASELINE_SEED.
def script(task, ctx):
    import os
    from yam_arena.demo import grip_rotation
    rng = np.random.default_rng(int(os.environ.get("BASELINE_SEED", "0")))
    while True:
        arm = ctx.left if rng.random() < 0.5 else ctx.right
        side = 1 if arm is ctx.left else -1
        x = rng.uniform(0.15, 0.55)
        y = rng.uniform(-0.05, 0.30) * side
        z = task.TOP + (rng.uniform(0.01, 0.03) if rng.random() < 0.4 else rng.uniform(0.03, 0.25))
        rot = grip_rotation(float(rng.uniform(-math.pi, math.pi)))
        yield from arm.move(np.array([x, y, z + 0.08]), rot, float(rng.uniform(0.8, 2.0)), grip=float(rng.uniform(0, 1)))
        yield from arm.move(np.array([x, y, z]), rot, 0.8)
        if rng.random() < 0.5:
            yield from arm.set_grip(0.0, 0.6)
            yield from arm.move(np.array([x + rng.uniform(-0.1, 0.1), y + rng.uniform(-0.1, 0.1), z + 0.1]), rot, 1.0)
            yield from arm.set_grip(1.0, 0.4)
        yield from wait(float(rng.uniform(0, 2)))
        yield from arm.move(np.array([0.2, 0.25 * side, task.TOP + 0.25]), rot, 1.0)
